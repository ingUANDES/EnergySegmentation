\begin{frame}{Modelo CAP-and-Trade de  Amigo, Cea-Echenique \& Feijoo (2021)}
\small
    \begin{itemize}
        \item Modelo en dos etapas para el sector eléctrico chileno.
        \item Etapa 1: el planificador fija la cantidad total de permisos $\theta$, productores deciden permisos a comprar, generación de electricidad e inversión en capacidad extra, todo con incertidumbre en la demanda futura.
        \item Etapa 2: los generadores ajustan inversión y producción bajo un límite de emisiones. Se encuentran en dos mercados: el de electricidad y el de permisos transables.
        \item El objetivo es evaluar cómo distintos límites de $\theta (tCO_2e)$ permiten converger a neutralidad al 2050.
    \end{itemize}
\end{frame}


\begin{frame}{Planificador Social  Amigo, Cea-Echenique \& Feijoo (2021): función objetivo}
\small
El planificador elige $\theta$ considerando incertidumbre en el verdadero límite ambiental
\[
CAP \sim \mathcal{N}(\mu,\sigma^2)
\]

y debe asegurar que no lo exceda con alta probabilidad:
\[
\Pr(\theta \ge CAP) \le \epsilon
\]

\textbf{Problema a maximizar:}
\[
\max_{\theta} \ \theta\,\pi^a - \mathcal{F}(\theta)
\]

sujeto a la restricción probabilística:
\[
\varphi^{-1}(\epsilon)\,\sigma + \mu - \theta \ge 0
\]
\end{frame}


\begin{frame}{Formulación del planificador con atención limitada}
\small
\begin{itemize}
    \item CAP incierto:
    \[
        CAP \sim \mathcal{N}(CAP_d,\sigma_{CAP}^2)
    \]
    \item Señal observada:
    \[
        CAP_s = CAP + \varepsilon
    \]
    \item Nivel de atención:
    \[
        m = \frac{\sigma_{CAP}^2}{\sigma_{CAP}^2 + \sigma_\varepsilon^2}
    \]
    \textit{donde } $m=1$ (perfecta) y $m=0$ ($CAP_d$).
\end{itemize}
\end{frame}
\begin{frame}{Atención limitada}
\small
\begin{itemize}
    \item Utilidad del planificador:
    \[
        \mathcal{V}(\theta,CAP) = \pi^a \theta - \frac{\kappa}{2}(\theta - CAP)^2.
    \]
    \item Se maximiza la utilidad condicionada en $CAP_S$:
    \[
        \max_{\theta} \ \mathbb{E}\!\left[\pi^a \theta - \frac{\kappa}{2}(\theta-CAP)^2 \,\bigg|\,CAP_s\right]
    \]
    \item se obtiene CPO:
    \[
        0=\mathbb{E}\!\left[\pi^a - \kappa(\theta - CAP)\,\bigg|\,CAP_s\right]
    \]

\end{itemize}
\end{frame}


\begin{frame}{Atención limitada}
\small
\begin{itemize}
    \item despejando:
    \[
        0 
        = \pi^a - \kappa\,\theta + \kappa\,\mathbb{E}[CAP \mid CAP_s]
    \]
        \item Se toma la ecuación de (Gabaix, 2019):
    \[
    \mathbb{E}\left[CAP|CAP_s\right] = m CAP_s+ (1 - m)CAP_d,
    \]
    \item Remplazando los valores la acción óptima del planificador esta dada por:
    \[
        \theta^*(CAP_s) = m\,CAP_s + (1-m)\,CAP_d + \frac{\pi^a}{\kappa}
    \]

\end{itemize}
\end{frame}

\begin{frame}{Contribución principal}
\begin{itemize}
    \item Atención al mecanismo de asignación de permisos.
    \item El CAP ya no es procesado perfectamente: el planificador encuentra una señal de base a la que se ancla y se ajusta según un nivel de atención a una señal más cercana al verdadero presupuesto.
    \item Permite analizar:
    \begin{itemize}
        \item Transición a tecnologías limpias en función de la atención.
        \item Comparación con presupuestos nacionales.
        \item Cambios en la utilidad del subastador con distintos niveles de atención.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Parámetros a Priorizar }
\begin{itemize}
    \item \textbf{Q\_bar(I)}: Capacidad máxima de operación [MW]
    \item \textbf{Inv(I)}: Costo de inversión [USD/kW]
    \item \textbf{C(I)}: Costo marginal de operación [USD/kWh]
    \item \textbf{epsilon(I)}: Factor de emisión [tCO\(_2\)e/MWh]
    \item \textbf{CF(I)}: Factor de capacidad
    \item \textbf{RP(I)}: Potencial máximo de recurso[MW]
    \item \textbf{TC(I,TAU2)}: Cambio tecnológico (operación) en el tiempo
    \item \textbf{TCR(I,TAU2)}: Cambio tecnológico recursivo (inversión)
    \item     \textbf{D1(TAU)}: Demanda en primera etapa
    \item \textbf{D2(W,TAU2)}: Demanda en segunda etapa por escenario

\end{itemize}
\end{frame}
